function resolveContrastColor(el: HTMLElement): "black" | "white" {
  const bg = getComputedStyle(el).backgroundColor;
  const canvas = document.createElement("canvas");
  canvas.width = 1;
  canvas.height = 1;
  const ctx = canvas.getContext("2d");
  if (!ctx) return "black";
  ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue("--background").trim() || "#ffffff";
  ctx.fillRect(0, 0, 1, 1);
  ctx.fillStyle = bg;
  ctx.fillRect(0, 0, 1, 1);
  const { data } = ctx.getImageData(0, 0, 1, 1);
  const r = data[0] ?? 0;
  const g = data[1] ?? 0;
  const b = data[2] ?? 0;
  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255 > 0.5 ? "black" : "white";
}

function ColorItem({ text, bgColor }: { text: string; bgColor: string }): ReactElement {
  const ref = useRef<HTMLDivElement>(null);
  const [textColor, setTextColor] = useState<"black" | "white">("black");

  useEffect(() => {
    function update(): void {
      if (ref.current) setTextColor(resolveContrastColor(ref.current));
    }
    update();
    const observer = new MutationObserver(update);
    observer.observe(document.documentElement, { attributes: true, attributeFilter: ["class", "data-theme"] });
    return (): void => observer.disconnect();
  }, [bgColor]);

  return (
    <Card variant="secondary" ref={ref} className={`m-2 h-15 w-35 ${bgColor}`}>
      <CardContent className="items-center justify-center truncate text-xs" style={{ color: textColor }}>
        {text}
      </CardContent>
    </Card>
  );
}

  const context = findContext(DEFAULT_CONTEXTS, parsed.context)!;
  const preview: QuickAddInputProps["preview"] = value.trim()
    ? {
        title: parsed.title,
        contextName: context.name,
        contextColor: context.color,
        due: parsed.due,
        remindAt: parsed.remindAt,
        important: parsed.important,
        tags: parsed.tags.filter((tag) => tag !== context.tag),
        links: parsed.links,
      }
    : null;

  return (
    <div className="w-[640px] max-w-full">
      <QuickAddInput
        value={value}
        onChange={setValue}
        onSubmit={() => {
          onSubmit(value);
          setValue("");

        today={TODAY}

  );
}

function resolveContrastColor(el: HTMLElement): "black" | "white" {
  const bg = getComputedStyle(el).backgroundColor;
  const canvas = document.createElement("canvas");
  canvas.width = 1;
  canvas.height = 1;
  const ctx = canvas.getContext("2d");
  if (!ctx) return "black";
  ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue("--background").trim() || "#ffffff";
  ctx.fillRect(0, 0, 1, 1);
  ctx.fillStyle = bg;
  ctx.fillRect(0, 0, 1, 1);
  const { data } = ctx.getImageData(0, 0, 1, 1);
  const r = data[0] ?? 0;
  const g = data[1] ?? 0;
  const b = data[2] ?? 0;
  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255 > 0.5 ? "black" : "white";
}

function ColorItem({ text, bgColor }: { text: string; bgColor: string }): ReactElement {
  const ref = useRef<HTMLDivElement>(null);
  const [textColor, setTextColor] = useState<"black" | "white">("black");

  useEffect(() => {
    function update(): void {
      if (ref.current) setTextColor(resolveContrastColor(ref.current));
    }
    update();
    const observer = new MutationObserver(update);
    observer.observe(document.documentElement, { attributes: true, attributeFilter: ["class", "data-theme"] });
    return (): void => observer.disconnect();
  }, [bgColor]);

  return (
    <Card variant="secondary" ref={ref} className={`m-2 h-15 w-35 ${bgColor}`}>
      <CardContent className="items-center justify-center truncate text-xs" style={{ color: textColor }}>
        {text}
      </CardContent>
    </Card>
  );
}
